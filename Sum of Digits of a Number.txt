class Intprop {

    int sumdigit(int n) {
        int ans = 0;

        while (n > 0) {
            ans = ans + n % 10;
            n = n / 10;
        }

        return ans;
    }
}

public class Main {

    public static void main(String[] args) {

        Intprop obj = new Intprop();

        int ans = obj.sumdigit(874511);

        System.out.println(ans);
    }
}
